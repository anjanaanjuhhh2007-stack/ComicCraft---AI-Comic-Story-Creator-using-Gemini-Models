 ### 3\. Create `app/api/generate/route.ts`

```
import { NextRequest, NextResponse } from "next/server";
import { GoogleGenAI } from "@google/genai";

const ai = new GoogleGenAI({
  apiKey: process.env.GEMINI_API_KEY,
});

export async function POST(request: NextRequest) {
  try {
    const {
      idea,
      genre = "Adventure",
      tone = "Fun",
      pages = 5,
    } = await request.json();

    if (!idea || typeof idea !== "string") {
      return NextResponse.json(
        { error: "Story idea is required." },
        { status: 400 }
      );
    }

    const prompt = `
You are ComicCraft, an AI comic story creator.

Create an original comic from this idea:

"${idea}"

Genre: ${genre}
Tone: ${tone}
Number of pages: ${pages}

Create 2-5 main characters.

Create exactly ${pages} pages.

Create 3 or 4 panels per page.

The story must have:
- Beginning
- Conflict
- Development
- Climax
- Resolution

Maintain character consistency across all pages.

Return ONLY valid JSON.

Use exactly this structure:

{
  "title": "string",
  "genre": "string",
  "tone": "string",
  "summary": "string",
  "characters": [
    {
      "name": "string",
      "role": "string",
      "personality": "string",
      "appearance": "string"
    }
  ],
  "pages": [
    {
      "pageNumber": 1,
      "title": "string",
      "panels": [
        {
          "panelNumber": 1,
          "sceneDescription": "string",
          "caption": "string",
          "visualPrompt": "string",
          "dialogue": [
            {
              "character": "string",
              "text": "string"
            }
          ]
        }
      ]
    }
  ]
}
`;

    const response = await ai.models.generateContent({
      model: "gemini-2.5-flash",
      contents: prompt,
      config: {
        responseMimeType: "application/json",
      },
    });

    const text = response.text;

    if (!text) {
      return NextResponse.json(
        { error: "Gemini returned an empty response." },
        { status: 500 }
      );
    }

    let comic;

    try {
      comic = JSON.parse(text);
    } catch {
      return NextResponse.json(
        { error: "Gemini returned invalid JSON." },
        { status: 500 }
      );
    }

    return NextResponse.json(comic);
  } catch (error) {
    console.error("Comic generation error:", error);

    return NextResponse.json(
      {
        error:
          "Unable to generate the comic. Please try again.",
      },
      { status: 500 }
    );
  }
}
```
